cmake_minimum_required(VERSION 3.22)
project(blindpilot_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.deps/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run tools/setup-native.ps1 to fetch the pinned llama.cpp source.")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT LLAMA_REV STREQUAL "3bb63cbcbfb62056c84fe9e521b0f3d756d29c32")
    message(FATAL_ERROR "llama.cpp revision differs from the tested pin. Run tools/setup-native.ps1.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(FLAG GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI GGML_BACKEND_DL LLAMA_OPENSSL LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_CONNECT)
    set(${FLAG} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(blindpilot_llama SHARED local-runtime.cpp)
target_link_libraries(blindpilot_llama PRIVATE llama log android)
target_link_options(blindpilot_llama PRIVATE "-Wl,-z,max-page-size=16384")
